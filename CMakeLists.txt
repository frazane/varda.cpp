cmake_minimum_required(VERSION 3.24)
project(varda LANGUAGES CXX)

option(VARDA_CUDA "Build the CUDA backend" ON)
option(VARDA_NATIVE "Tune CPU code for the build host (-march=native)" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Executables find cuBLAS/OpenBLAS from the toolchain paths without LD_LIBRARY_PATH.
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH ON)
set(CMAKE_BUILD_RPATH_USE_LINK_PATH ON)

find_package(OpenMP REQUIRED)
set(BLA_VENDOR OpenBLAS)
find_package(BLAS REQUIRED)
find_path(CBLAS_INCLUDE_DIR cblas.h HINTS ENV OPENBLAS_HOME PATH_SUFFIXES include include/openblas REQUIRED)

# CPU numerics: no implicit FMA contraction, so element-wise code rounds like the
# reference's separate ops. Hot loops that want FMA use it explicitly.
set(VARDA_CXX_FLAGS -ffp-contract=off -Wall -Wextra -Wno-unused-parameter)
if(VARDA_NATIVE)
  list(APPEND VARDA_CXX_FLAGS -march=native)
else()
  list(APPEND VARDA_CXX_FLAGS -mavx2 -mfma -mf16c)
endif()

file(GLOB VARDA_CORE_SRC CONFIGURE_DEPENDS src/varda/*.cpp src/varda/cpu/*.cpp)
add_library(varda STATIC ${VARDA_CORE_SRC})
target_include_directories(varda PUBLIC src third_party ${CBLAS_INCLUDE_DIR})
target_compile_options(varda PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${VARDA_CXX_FLAGS}>)
target_link_libraries(varda PUBLIC OpenMP::OpenMP_CXX ${BLAS_LIBRARIES})

if(VARDA_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    if(NOT CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES 80 CACHE STRING "CUDA architectures (A100 = 80)" FORCE)
    endif()
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    find_package(CUDAToolkit REQUIRED)
    file(GLOB VARDA_CUDA_SRC CONFIGURE_DEPENDS src/varda/cuda/*.cu src/varda/cuda/*.cpp)
    target_sources(varda PRIVATE ${VARDA_CUDA_SRC})
    target_compile_definitions(varda PUBLIC VARDA_WITH_CUDA)
    target_compile_options(varda PRIVATE
      $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -lineinfo -Xcompiler=-Wall>)
    target_link_libraries(varda PUBLIC CUDA::cudart CUDA::cublas CUDA::cublasLt)
  else()
    message(WARNING "VARDA_CUDA=ON but no CUDA compiler found: building the CPU backend only")
  endif()
endif()

file(GLOB VARDA_APPS CONFIGURE_DEPENDS apps/*.cpp)
foreach(app ${VARDA_APPS})
  get_filename_component(name ${app} NAME_WE)
  add_executable(${name} ${app})
  target_compile_options(${name} PRIVATE ${VARDA_CXX_FLAGS})
  target_compile_definitions(${name} PRIVATE VARDA_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  target_link_libraries(${name} PRIVATE varda)
endforeach()

enable_testing()
file(GLOB VARDA_TESTS CONFIGURE_DEPENDS tests/*.cpp tests/*.cu)
foreach(t ${VARDA_TESTS})
  get_filename_component(name ${t} NAME_WE)
  get_filename_component(ext ${t} EXT)
  if(ext STREQUAL ".cu" AND NOT CMAKE_CUDA_COMPILER)
    continue()
  endif()
  add_executable(${name} ${t})
  target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${VARDA_CXX_FLAGS}>)
  target_link_libraries(${name} PRIVATE varda)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
