cmake_minimum_required(VERSION 3.24)
project(varda LANGUAGES CXX)

option(VARDA_CUDA "Build the CUDA backend" ON)
option(VARDA_NATIVE "Tune CPU code for the build host (-march=native)" OFF)
option(VARDA_GRIB "Build the GRIB input/output layer (needs ecCodes, ECCODES_HOME)" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Executables find cuBLAS/OpenBLAS from the toolchain paths without LD_LIBRARY_PATH.
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH ON)
set(CMAKE_BUILD_RPATH_USE_LINK_PATH ON)

find_package(OpenMP REQUIRED)
set(BLA_VENDOR OpenBLAS)
find_package(BLAS REQUIRED)
find_path(CBLAS_INCLUDE_DIR cblas.h HINTS ENV OPENBLAS_HOME PATH_SUFFIXES include include/openblas REQUIRED)

# CPU numerics: no implicit FMA contraction, so element-wise code rounds like the
# reference's separate ops. Hot loops that want FMA use it explicitly.
set(VARDA_CXX_FLAGS -ffp-contract=off -Wall -Wextra -Wno-unused-parameter)
if(VARDA_NATIVE)
  list(APPEND VARDA_CXX_FLAGS -march=native)
else()
  list(APPEND VARDA_CXX_FLAGS -mavx2 -mfma -mf16c)
endif()

file(GLOB VARDA_CORE_SRC CONFIGURE_DEPENDS src/varda/*.cpp src/varda/cpu/*.cpp)
add_library(varda STATIC ${VARDA_CORE_SRC})
target_include_directories(varda PUBLIC src third_party ${CBLAS_INCLUDE_DIR})
target_compile_options(varda PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${VARDA_CXX_FLAGS}>)
target_link_libraries(varda PUBLIC OpenMP::OpenMP_CXX ${BLAS_LIBRARIES})

if(VARDA_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    if(NOT CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES 80)  # A100; override with -DCMAKE_CUDA_ARCHITECTURES=...
    endif()
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    find_package(CUDAToolkit REQUIRED)
    file(GLOB VARDA_CUDA_SRC CONFIGURE_DEPENDS src/varda/cuda/*.cu src/varda/cuda/*.cpp)
    target_sources(varda PRIVATE ${VARDA_CUDA_SRC})
    # varda was created before enable_language(CUDA), so it did not pick up the default.
    set_target_properties(varda PROPERTIES CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
    target_compile_definitions(varda PUBLIC VARDA_WITH_CUDA)
    target_compile_options(varda PRIVATE
      $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -lineinfo -Xcompiler=-Wall>)
    target_link_libraries(varda PUBLIC CUDA::cudart CUDA::cublas CUDA::cublasLt)
  else()
    message(WARNING "VARDA_CUDA=ON but no CUDA compiler found: building the CPU backend only")
  endif()
endif()

# GRIB input/output: a separate library so the engine core stays free of ecCodes.
# ECCODES_HOME holds eccodeslib/ and eckitlib/ (the layout of the eccodeslib and eckitlib
# wheels, the same build the reference anemoi-inference uses; tools/env.sh).
set(VARDA_HAVE_GRIB OFF)
if(VARDA_GRIB)
  find_path(ECCODES_INCLUDE_DIR eccodes.h HINTS ENV ECCODES_HOME PATH_SUFFIXES eccodeslib/include include)
  find_library(ECCODES_LIBRARY eccodes HINTS ENV ECCODES_HOME PATH_SUFFIXES eccodeslib/lib64 lib64 lib)
  if(ECCODES_INCLUDE_DIR AND ECCODES_LIBRARY)
    set(VARDA_HAVE_GRIB ON)
    get_filename_component(ECCODES_LIBDIR ${ECCODES_LIBRARY} DIRECTORY)
    file(GLOB VARDA_IO_SRC CONFIGURE_DEPENDS src/varda/io/*.cpp)
    add_library(varda_io STATIC ${VARDA_IO_SRC})
    target_include_directories(varda_io PUBLIC ${ECCODES_INCLUDE_DIR})
    target_compile_options(varda_io PRIVATE ${VARDA_CXX_FLAGS})
    # libeccodes needs eckit next to it; old-style RPATH reaches transitive dependencies.
    target_link_libraries(varda_io PUBLIC varda ${ECCODES_LIBRARY}
      "-Wl,--disable-new-dtags,-rpath,${ECCODES_LIBDIR}:${ECCODES_LIBDIR}/../../eckitlib/lib64")
  else()
    message(WARNING "VARDA_GRIB=ON but ecCodes not found (set ECCODES_HOME): no GRIB layer")
  endif()
endif()

file(GLOB VARDA_APPS CONFIGURE_DEPENDS apps/*.cpp)
if(VARDA_HAVE_GRIB)
  file(GLOB VARDA_IO_APPS CONFIGURE_DEPENDS apps/io/*.cpp)
endif()
foreach(app ${VARDA_APPS})
  get_filename_component(name ${app} NAME_WE)
  add_executable(${name} ${app})
  target_compile_options(${name} PRIVATE ${VARDA_CXX_FLAGS})
  target_compile_definitions(${name} PRIVATE VARDA_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  target_link_libraries(${name} PRIVATE varda)
endforeach()
foreach(app ${VARDA_IO_APPS})
  get_filename_component(name ${app} NAME_WE)
  add_executable(${name} ${app})
  target_include_directories(${name} PRIVATE apps)
  target_compile_options(${name} PRIVATE ${VARDA_CXX_FLAGS})
  target_compile_definitions(${name} PRIVATE VARDA_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  target_link_libraries(${name} PRIVATE varda_io)
endforeach()

enable_testing()
file(GLOB VARDA_TESTS CONFIGURE_DEPENDS tests/*.cpp tests/*.cu)
foreach(t ${VARDA_TESTS})
  get_filename_component(name ${t} NAME_WE)
  get_filename_component(ext ${t} EXT)
  if(ext STREQUAL ".cu" AND NOT CMAKE_CUDA_COMPILER)
    continue()
  endif()
  add_executable(${name} ${t})
  target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${VARDA_CXX_FLAGS}>)
  target_link_libraries(${name} PRIVATE varda)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
if(VARDA_HAVE_GRIB)
  file(GLOB VARDA_IO_TESTS CONFIGURE_DEPENDS tests/io/*.cpp)
  foreach(t ${VARDA_IO_TESTS})
    get_filename_component(name ${t} NAME_WE)
    add_executable(${name} ${t})
    target_compile_options(${name} PRIVATE ${VARDA_CXX_FLAGS})
    target_link_libraries(${name} PRIVATE varda_io)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
